/**
 * Initials for the circular logo, used for a business that has not uploaded one.
 */
export function initialsOf(name: string): string {
  const words = name
    .split(/\s+/)
    .map((word) => word.replace(/[^A-Za-z0-9]/g, ""))
    .filter(Boolean);
  if (words.length === 0) return "?";
  if (words.length === 1) return words[0]!.slice(0, 2).toUpperCase();
  return words
    .slice(0, 3)
    .map((word) => word.charAt(0))
    .join("")
    .toUpperCase();
}

export type ApiFailure = { ok: false; status: number; message: string; fieldErrors: Record<string, string> };
export type ApiResult<T> = { ok: true; data: T } | ApiFailure;

async function toFailure(response: Response): Promise<ApiFailure> {
  let message = "Something went wrong. Please try again.";
  const fieldErrors: Record<string, string> = {};
  try {
    const body = (await response.json()) as { detail?: unknown; errors?: Record<string, unknown> };
    if (body.errors && typeof body.errors === "object") {
      for (const [field, messages] of Object.entries(body.errors)) {
        if (Array.isArray(messages) && typeof messages[0] === "string") fieldErrors[field] = messages[0];
      }
      message = "Please fix the fields marked below.";
    } else if (typeof body.detail === "string") {
      message = body.detail;
    }
  } catch {
    // Not JSON: keep the generic message.
  }
  return { ok: false, status: response.status, message, fieldErrors };
}

async function send<T>(path: string, init: RequestInit, accessToken?: string): Promise<ApiResult<T>> {
  const headers = new Headers(init.headers);
  headers.set("Accept", "application/json");
  if (init.body) headers.set("Content-Type", "application/json");
  if (accessToken) headers.set("Authorization", `Bearer ${accessToken}`);
  let response: Response;
  try {
    response = await fetch(`${API_BASE_URL}${path}`, { ...init, headers, cache: "no-store" });
  } catch {
    return { ok: false, status: 0, message: "We could not reach the server. Please try again in a moment.", fieldErrors: {} };
  }
  if (!response.ok) return toFailure(response);
  if (response.status === 204) return { ok: true, data: undefined as T };
  return { ok: true, data: (await response.json()) as T };
}

/** Calls an owner endpoint with the signed-in owner's token. No token or a rejected token sends the person to sign in again. */
async function authed<T>(path: string, init: RequestInit = {}): Promise<ApiResult<T>> {
  const { access } = await readTokens();
  if (!access) redirect("/login/reset");
  const result = await send<T>(path, init, access);
  if (!result.ok && result.status === 401) redirect("/login/reset");
  return result;
}

using BusinessDirectory.Api.Data;
using System.Security.Claims;
using BusinessDirectory.Api.Media;
using Microsoft.Data.SqlClient;

namespace BusinessDirectory.Api.Endpoints;

/// <summary>
/// A listing's photos, logo and video, under /api/v1/me/listings/{id}/media.
///
/// Three things are enforced here and nowhere else:
///   * ownership â€” every statement joins business_members, so another owner's listing is "not found";
///   * the plan's allowance â€” counted inside the same transaction that inserts the row, so two
///     uploads racing cannot both slip past the last free slot;
///   * what the file actually is â€” decided from its first bytes, never from the name or the
///     Content-Type the caller sent.
/// </summary>
public static class OwnerMediaEndpoints
{
    public sealed record MediaItemDto(
        long Id,
        string Kind,
        string Url,
        string ContentType,
        long ByteSize,
        int? Width,
        int? Height,
        int SortOrder,
        /// <summary>D10: over-allowance media after a downgrade. Kept, not deleted, and not shown publicly.</summary>
        bool IsHidden,
        DateTimeOffset CreatedAt);

    /// <param name="LimitsConfirmed">
    /// False while the per-tier allowances are still provisional (see V0008). The owner screens say so
    /// rather than presenting an unconfirmed number as a promise.
    /// </param>
    /// <param name="UploadsAvailable">
    /// False when no storage is configured; the UI then explains it instead of offering a button that cannot work.
    /// </param>
    public sealed record MediaAllowanceDto(
        string PlanSlug,
        string PlanName,
        int PhotoLimit,
        int VideoLimit,
        int LogoLimit,
        int PhotosUsed,
        int VideosUsed,
        int LogosUsed,
        bool LimitsConfirmed,
        bool UploadsAvailable);

    public sealed record MediaDto(MediaAllowanceDto Allowance, List<MediaItemDto> Items);

    public static RouteGroupBuilder MapOwnerMediaEndpoints(this RouteGroupBuilder listings)
    {
        var group = listings.MapGroup("/{id:long}/media");
        group.MapGet("", List);
        // The form is read from HttpRequest by hand, which also keeps the endpoint clear of minimal-API
        // antiforgery metadata: this API is authenticated by a bearer token, not by a cookie, so a
        // cross-site form post cannot carry the caller's credentials in the first place.
        group.MapPost("", Upload);
        group.MapPost("/{mediaId:long}/move", Move);
        group.MapDelete("/{mediaId:long}", Delete);
        return group;
    }

    // ------------------------------------------------------------------ list
    private static async Task<IResult> List(
        long id, ClaimsPrincipal principal, SqlDatabase db, IMediaStorage storage, CancellationToken ct)
    {
        if (!MeEndpoints.TryGetUserId(principal, out var userId))
        {
            return Results.Unauthorized();
        }

        await using var conn = await db.OpenConnectionAsync(ct);
        var allowance = await ReadAllowance(conn, null, userId, id, storage.IsConfigured, ct);
        if (allowance is null)
        {
            return Results.NotFound();
        }

        var items = await ReadItems(conn, id, storage, ct);
        return Results.Ok(new MediaDto(allowance, items));
    }

    // ------------------------------------------------------------------ upload
    private static async Task<IResult> Upload(
        long id,
        HttpRequest request,
        ClaimsPrincipal principal,
        SqlDatabase db,
        IMediaStorage storage,
        ILogger<OwnerMediaUpload> logger,
        CancellationToken ct)
    {
        if (!MeEndpoints.TryGetUserId(principal, out var userId))
        {
            return Results.Unauthorized();
        }

        if (!storage.IsConfigured)
        {
            // Nothing is written to the database either: a row pointing at bytes that were never
            // stored would show as a permanently broken image.
            return Results.Problem("Uploads are not available right now.", statusCode: StatusCodes.Status503ServiceUnavailable);
        }

        if (!request.HasFormContentType)
        {
            return Results.Problem("Send the file as multipart/form-data.", statusCode: StatusCodes.Status415UnsupportedMediaType);
        }

        var form = await request.ReadFormAsync(ct);
        if (!Enum.TryParse<MediaKind>(form["kind"], ignoreCase: true, out var kind))
        {
            return Results.ValidationProblem(new Dictionary<string, string[]> { ["kind"] = ["Choose logo, photo or video."] });
        }

        var file = form.Files["file"];
        if (file is null || file.Length == 0)
        {
            return Results.ValidationProblem(new Dictionary<string, string[]> { ["file"] = ["Choose a file."] });
        }

        var maxBytes = MediaFile.MaxBytesFor(kind);
        if (file.Length > maxBytes)
        {
            return Results.Problem(
                $"That file is {file.Length / (1024 * 1024)} MB. The limit is {maxBytes / (1024 * 1024)} MB.",
                statusCode: StatusCodes.Status413PayloadTooLarge);
        }

        // Enough of the start of the file to identify it and, for an image, read its dimensions.
        var head = new byte[64 * 1024];
        int headLength;
        await using (var peek = file.OpenReadStream())
        {
            headLength = await peek.ReadAtLeastAsync(
                head, Math.Min(head.Length, (int)file.Length), throwOnEndOfStream: false, ct);
        }

        var contentType = MediaFile.SniffContentType(head.AsSpan(0, headLength));
        var matchesKind = kind == MediaKind.Video
            ? contentType == MediaFile.Mp4
            : MediaFile.ImageTypes.Contains(contentType);
        if (contentType is null || !matchesKind)
        {
            return Results.Problem(
                kind == MediaKind.Video ? "Videos must be MP4." : "Photos must be JPEG, PNG or WebP.",
                statusCode: StatusCodes.Status415UnsupportedMediaType);
        }

        var size = kind == MediaKind.Video ? null : MediaFile.ReadImageSize(head.AsSpan(0, headLength), contentType);
        var storageKey = MediaFile.NewStorageKey(id, kind, contentType);

        await using var conn = await db.OpenConnectionAsync(ct);
        // The whole check-and-insert runs in one transaction, and the listing row is locked first, so
        // the count a concurrent upload sees already includes this one.
        await using var tx = await conn.BeginSqlTransactionAsync(ct);

        var allowance = await ReadAllowance(conn, tx, userId, id, uploadsAvailable: true, ct);
        if (allowance is null)
        {
            return Results.NotFound();
        }

        var (limit, used) = kind switch
        {
            MediaKind.Logo => (allowance.LogoLimit, allowance.LogosUsed),
            MediaKind.Photo => (allowance.PhotoLimit, allowance.PhotosUsed),
            _ => (allowance.VideoLimit, allowance.VideosUsed),
        };
        if (used >= limit)
        {
            var what = kind.ToString().ToLowerInvariant();
            return Results.Problem(
                limit == 0
                    ? $"The {allowance.PlanName} plan does not include {what}s."
                    : $"The {allowance.PlanName} plan allows {limit} {what}{(limit == 1 ? "" : "s")}. Remove one first.",
                statusCode: StatusCodes.Status409Conflict);
        }

        long mediaId;
        await using (var insert = new SqlCommand(
            @"INSERT INTO listing_media (listing_id, kind, storage_key, content_type, byte_size, width, height, sort_order, created_by_user_id)
              OUTPUT inserted.id
              VALUES (@listing, @kind, @key, @type, @size, @width, @height,
                      COALESCE((SELECT max(sort_order) + 1 FROM listing_media WHERE listing_id = @listing AND kind = @kind), 0),
                      @user)", conn, tx))
        {
            insert.Parameters.AddWithValue("listing", id);
            insert.Parameters.AddWithValue("kind", kind.ToString().ToLowerInvariant());
            insert.Parameters.AddWithValue("key", storageKey);
            insert.Parameters.AddWithValue("type", contentType);
            insert.Parameters.AddWithValue("size", file.Length);
            insert.Parameters.AddWithValue("width", (object?)size?.Width ?? DBNull.Value);
            insert.Parameters.AddWithValue("height", (object?)size?.Height ?? DBNull.Value);
            insert.Parameters.AddWithValue("user", userId);
            mediaId = (long)(await insert.ExecuteScalarAsync(ct))!;
        }

        // The bytes are stored only after the row is reserved, and the row is committed only once the
        // bytes are safely there. A failure in between rolls the row back, so the gallery never lists
        // a file that is not in storage.
        try
        {
            await using var content = file.OpenReadStream();
            await storage.SaveAsync(storageKey, contentType, content, ct);
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Upload to media storage failed for listing {ListingId}.", id);
            await tx.RollbackAsync(CancellationToken.None);
            return Results.Problem("The file could not be stored. Please try again.", statusCode: StatusCodes.Status502BadGateway);
        }

        try
        {
            await tx.CommitAsync(ct);
        }
        catch (Exception)
        {
            // The bytes are stored but the row is gone, so nothing would ever reference them again.
            await storage.DeleteAsync(storageKey, CancellationToken.None);
            throw;
        }

        return Results.Created($"/api/v1/me/listings/{id}/media/{mediaId}", new MediaItemDto(
            mediaId, kind.ToString().ToLowerInvariant(), storage.PublicUrl(storageKey), contentType,
            file.Length, size?.Width, size?.Height, 0, false, DateTimeOffset.UtcNow));
    }

    // ------------------------------------------------------------------ reorder
    /// <summary>
    /// Moves one photo one place earlier or later, by swapping its position with its neighbour of
    /// the same kind. A swap rather than a renumbering: it touches two rows, so two owners reordering
    /// at once cannot leave the gallery in a half-renumbered state.
    /// </summary>
    private static async Task<IResult> Move(
        long id, long mediaId, string? direction, ClaimsPrincipal principal, SqlDatabase db, CancellationToken ct)
    {
        if (!MeEndpoints.TryGetUserId(principal, out var userId))
        {
            return Results.Unauthorized();
        }

        var earlier = direction switch
        {
            "up" => true,
            "down" => false,
            _ => (bool?)null,
        };
        if (earlier is null)
        {
            return Results.ValidationProblem(new Dictionary<string, string[]> { ["direction"] = ["Send up or down."] });
        }

        await using var conn = await db.OpenConnectionAsync(ct);
        await using var tx = await conn.BeginSqlTransactionAsync(ct);

        // The business_members join is the authorisation, exactly as in the other handlers.
        string kind;
        int sortOrder;
        await using (var find = new SqlCommand(
            /* UPDLOCK on the media row only, which is what FOR NO KEY UPDATE OF m said: the
               listing and the membership are read, not changed. */
            @"SELECT m.kind, m.sort_order
              FROM listing_media m WITH (UPDLOCK, ROWLOCK)
              JOIN listings li ON li.id = m.listing_id
              JOIN business_members bm ON bm.business_id = li.business_id AND bm.user_id = @user
              WHERE m.id = @media AND m.listing_id = @listing", conn, tx))
        {
            find.Parameters.AddWithValue("user", userId);
            find.Parameters.AddWithValue("media", mediaId);
            find.Parameters.AddWithValue("listing", id);
            await using var reader = await find.ExecuteReaderAsync(ct);
            if (!await reader.ReadAsync(ct))
            {
                return Results.NotFound();
            }
            kind = reader.GetString(0);
            sortOrder = reader.GetInt32(1);
        }

        long? neighbourId = null;
        var neighbourOrder = 0;
        await using (var neighbour = new SqlCommand(
            /* "the row just before this one, in (sort_order, id) order". PostgreSQL could compare
               the pair directly -- (sort_order, id) < (@order, @media) -- which SQL Server cannot, so
               it is written out: a smaller sort_order, or the same one and a smaller id. */
            $@"SELECT TOP (1) id, sort_order
               FROM listing_media WITH (UPDLOCK, ROWLOCK)
               WHERE listing_id = @listing AND kind = @kind
                 AND (sort_order {(earlier.Value ? "<" : ">")} @order
                      OR (sort_order = @order AND id {(earlier.Value ? "<" : ">")} @media))
               ORDER BY sort_order {(earlier.Value ? "DESC" : "ASC")}, id {(earlier.Value ? "DESC" : "ASC")}", conn, tx))
        {
            neighbour.Parameters.AddWithValue("listing", id);
            neighbour.Parameters.AddWithValue("kind", kind);
            neighbour.Parameters.AddWithValue("order", sortOrder);
            neighbour.Parameters.AddWithValue("media", mediaId);
            await using var reader = await neighbour.ExecuteReaderAsync(ct);
            if (await reader.ReadAsync(ct))
            {
                neighbourId = reader.GetInt64(0);
                neighbourOrder = reader.GetInt32(1);
            }
        }

        if (neighbourId is null)
        {
            // Already first or last. Not an error: the button simply had nothing to do. The rollback
            // waits until here because the reader above has to be closed before the connection is
            // free to carry another statement.
            await tx.RollbackAsync(ct);
            return Results.NoContent();
        }

        // Equal sort_order values can exist (a tie broken by id), and swapping those would change
        // nothing, so the moved row is pushed past its neighbour instead.
        var (mine, theirs) = sortOrder == neighbourOrder
            ? (earlier.Value ? neighbourOrder - 1 : neighbourOrder + 1, neighbourOrder)
            : (neighbourOrder, sortOrder);

        await using (var update = new SqlCommand(
            @"UPDATE listing_media SET sort_order = CASE id WHEN @media THEN @mine ELSE @theirs END
              WHERE id IN (@media, @neighbour)", conn, tx))
        {
            update.Parameters.AddWithValue("media", mediaId);
            update.Parameters.AddWithValue("neighbour", neighbourId.Value);
            update.Parameters.AddWithValue("mine", mine);
            update.Parameters.AddWithValue("theirs", theirs);
            await update.ExecuteNonQueryAsync(ct);
        }

        await tx.CommitAsync(ct);
        return Results.NoContent();
    }

    // ------------------------------------------------------------------ delete
    private static async Task<IResult> Delete(
        long id, long mediaId, ClaimsPrincipal principal, SqlDatabase db, IMediaStorage storage, CancellationToken ct)
    {
        if (!MeEndpoints.TryGetUserId(principal, out var userId))
        {
            return Results.Unauthorized();
        }

        // The join is the authorisation: a media id belonging to someone else's listing deletes nothing.
        await using var command = db.CreateCommand(
            /* DELETE ... USING becomes "DELETE <alias> FROM ... JOIN ...", and the row being
               removed is "deleted", not "inserted" -- an OUTPUT on a DELETE has no inserted row. */
            @"DELETE m
              OUTPUT deleted.storage_key
              FROM listing_media m
              JOIN listings li ON li.id = m.listing_id
              JOIN business_members bm ON bm.business_id = li.business_id AND bm.user_id = @user
              WHERE m.id = @media AND m.listing_id = @listing");
        command.Parameters.AddWithValue("user", userId);
        command.Parameters.AddWithValue("media", mediaId);
        command.Parameters.AddWithValue("listing", id);

        var storageKey = (string?)await command.ExecuteScalarAsync(ct);
        if (storageKey is null)
        {
            return Results.NotFound();
        }

        // The row is already gone. If removing the bytes fails, the only consequence is a blob nobody
        // references, which is better than a listing still showing a photo the owner deleted.
        if (storage.IsConfigured)
        {
            await storage.DeleteAsync(storageKey, ct);
        }
        return Results.NoContent();
    }

    // ------------------------------------------------------------------ shared reads
    /// <summary>
    /// The listing's plan and what it has used so far, or null when this user does not own the listing.
    /// Passed a transaction, it also locks the listing row, which is what makes the quota check reliable.
    /// FOR NO KEY UPDATE rather than FOR UPDATE: it blocks a second uploader without blocking the
    /// foreign keys that point at the listing (its media, enquiries and reviews).
    /// </summary>
    private static async Task<MediaAllowanceDto?> ReadAllowance(
        SqlConnection conn, SqlTransaction? tx, long userId, long listingId, bool uploadsAvailable, CancellationToken ct)
    {
        await using var command = new SqlCommand(
            @"SELECT p.slug, p.name, p.photo_limit, p.video_limit, p.logo_limit, p.media_limits_confirmed,
                     (SELECT COUNT_BIG(*) FROM listing_media m WHERE m.listing_id = li.id AND m.kind = 'photo' AND m.is_hidden = 0),
                     (SELECT COUNT_BIG(*) FROM listing_media m WHERE m.listing_id = li.id AND m.kind = 'video' AND m.is_hidden = 0),
                     (SELECT COUNT_BIG(*) FROM listing_media m WHERE m.listing_id = li.id AND m.kind = 'logo' AND m.is_hidden = 0)
              FROM listings li" + (tx is null ? "" : " WITH (UPDLOCK, ROWLOCK)") + @"
              JOIN business_members bm ON bm.business_id = li.business_id AND bm.user_id = @user
              JOIN plans p ON p.slug = li.plan_slug
              WHERE li.id = @listing AND li.status <> 'archived'",
            conn, tx);
        command.Parameters.AddWithValue("user", userId);
        command.Parameters.AddWithValue("listing", listingId);

        await using var reader = await command.ExecuteReaderAsync(ct);
        if (!await reader.ReadAsync(ct))
        {
            return null;
        }

        return new MediaAllowanceDto(
            reader.GetString(0),
            reader.GetString(1),
            reader.GetInt32(2),
            reader.GetInt32(3),
            reader.GetInt32(4),
            PhotosUsed: (int)reader.GetInt64(6),
            VideosUsed: (int)reader.GetInt64(7),
            LogosUsed: (int)reader.GetInt64(8),
            LimitsConfirmed: reader.GetBoolean(5),
            UploadsAvailable: uploadsAvailable);
    }

    private static async Task<List<MediaItemDto>> ReadItems(
        SqlConnection conn, long listingId, IMediaStorage storage, CancellationToken ct)

using BusinessDirectory.Api.Data;
using System.Data;
using BusinessDirectory.Api.Listings;
using BusinessDirectory.Api.Media;
using Microsoft.Data.SqlClient;

namespace BusinessDirectory.Api.Endpoints;

/// <summary>
/// Database-backed listing search under /api/v1/public/listings/search.
/// Matching uses SQL Server full-text search over name and description, the
/// hl_similarity function for typo tolerance, and the category_search_terms
/// table so a service word such as "tap repair" finds the right trade.
/// </summary>
public static class PublicSearchEndpoints
{
    public sealed record SearchListingDto(
        string Slug,
        string Name,
        string CitySlug,
        string LocalitySlug,
        string LocalityName,
        string CategorySlug,
        string CategoryName,
        int? DistanceMetres,
        string? AddressLine,
        string PhonePrimary,
        string? PhoneWhatsapp,
        string Url,
        /// <summary>The business's logo, or null when it has not uploaded one. The card falls back to a monogram.</summary>
        string? LogoUrl,
        /// <summary>Its first photo, used as the card thumbnail where there is no logo.</summary>
        string? PhotoUrl);

    public static RouteGroupBuilder MapPublicSearchEndpoints(this RouteGroupBuilder group)
    {
        group.MapGet("/listings/search", Search);
        return group;
    }

    // One statement does the filtering, ranking, distance, paging and total count.
    // {0} and {1} are the same ORDER BY twice, chosen from a fixed set below - never built from
    // request text. It is needed in both places: inside the CTE it decides *which* rows the page
    // holds, and on the final SELECT it decides what order they come back in. A CTE's own ordering
    // is not carried through the join below, so leaving the second one out returns the right rows
    // shuffled.
    //
    // The last SELECT starts from a single dummy row and LEFT JOINs the page, so it always returns
    // at least one row. That keeps the total available even when the requested page is past the end
    // (the listing columns are then NULL), which lets a client send the visitor back to page 1.
    /*
     * {0}/{1} are the two ORDER BYs described above. {2} and {3} are the text-matching half of the
     * query, which is present only when something was typed:
     *
     *   {2}  the joins -- two CONTAINSTABLE calls (full-text over name and over description) and the
     *        trigram similarity for typo tolerance
     *   {3}  the relevance expression built from them
     *   {4}  the WHERE clause that keeps only rows one of them matched
     *
     * They are composed rather than written inline because CONTAINSTABLE cannot be given a null or
     * empty condition: with nothing typed there is no full-text search to do, and the joins must not
     * be in the statement at all.
     */
    private const string SearchSqlFormat =
        @"WITH qry AS (
              SELECT CASE WHEN @lat IS NULL THEN NULL
                          ELSE geography::Point(@lat, @lng, 4326) END AS point
          ),
          matches AS (
              SELECT li.slug, li.name, ci.slug AS city_slug,
                     lo.slug AS locality_slug, lo.name AS locality_name,
                     pc.slug AS category_slug, pc.name AS category_name,
                     li.address_line, li.phone_primary, li.phone_whatsapp,
                     -- One logo and one photo per card. Hidden media (over a downgraded plan's
                     -- allowance, D10) is excluded here as well, so a card can never show a file the
                     -- listing page itself no longer displays.
                     (SELECT m.storage_key FROM listing_media m
                      WHERE m.listing_id = li.id AND m.kind = 'logo' AND m.is_hidden = 0
                      ORDER BY m.sort_order, m.id OFFSET 0 ROWS FETCH NEXT 1 ROWS ONLY) AS logo_key,
                     (SELECT m.storage_key FROM listing_media m
                      WHERE m.listing_id = li.id AND m.kind = 'photo' AND m.is_hidden = 0
                      ORDER BY m.sort_order, m.id OFFSET 0 ROWS FETCH NEXT 1 ROWS ONLY) AS photo_key,
                     li.location.STDistance(qry.point) AS distance_metres,
                     {3} AS relevance
              FROM qry
              JOIN listings li ON li.status = 'active'
              JOIN localities lo ON lo.id = li.locality_id
              JOIN cities ci ON ci.id = lo.city_id
              JOIN listing_categories lc ON lc.listing_id = li.id AND lc.is_primary = 1
              JOIN categories pc ON pc.id = lc.category_id
              {2}
              WHERE (@city IS NULL OR ci.slug = @city)
                -- Proximity window, used by locality landing pages so a thinly covered area
                -- still shows businesses just outside it. Uses the SQL Server spatial index on location.
                AND (@radius IS NULL OR li.location.STDistance(qry.point) <= @radius)
                -- A listing belongs to its own locality and to every locality it serves.
                AND (@locality IS NULL
                     OR lo.slug = @locality
                     OR EXISTS (SELECT 1
                                FROM listing_service_areas sa
                                JOIN localities sl ON sl.id = sa.locality_id
                                WHERE sa.listing_id = li.id AND sl.slug = @locality))
                -- A parent category slug also matches listings filed under its sub-categories.
                AND (@category IS NULL
                     OR EXISTS (SELECT 1
                                FROM listing_categories fc
                                JOIN categories fcat ON fcat.id = fc.category_id
                                LEFT JOIN categories fparent ON fparent.id = fcat.parent_id
                                WHERE fc.listing_id = li.id
                                  AND (fcat.slug = @category OR fparent.slug = @category)))
                {4}
          ),
          page AS (
              SELECT * FROM matches ORDER BY {0} OFFSET @offset ROWS FETCH NEXT @pageSize ROWS ONLY
          )
          SELECT (SELECT COUNT_BIG(*) FROM matches) AS total,
                 p.slug, p.name, p.city_slug, p.locality_slug, p.locality_name,
                 p.category_slug, p.category_name, p.address_line, p.phone_primary,
                 p.phone_whatsapp, p.distance_metres, p.logo_key, p.photo_key
          FROM (SELECT 1 AS one) AS always_one_row
          LEFT JOIN page p ON 1 = 1
          ORDER BY {1}";

    private static async Task<IResult> Search(
        string? q,
        string? city,
        string? locality,
        string? category,
        double? lat,
        double? lng,
        double? radiusKm,
        string? sort,
        int? page,
        int? pageSize,
        SqlDatabase db,
        IMediaStorage storage,
        CancellationToken ct)
    {
        var query = ListingSearchQuery.TryCreate(q, city, locality, category, lat, lng, radiusKm, sort, page, pageSize, out var error);
        if (query is null)
        {
            return Results.Problem(error, statusCode: StatusCodes.Status400BadRequest);
        }

        // The same ordering for the CTE (bare column names) and for the final SELECT (p.-qualified).
        var (pageOrder, resultOrder) = query.Sort switch
        {
            /* SQL Server has no NULLS LAST: it sorts nulls first ascending and last descending.
               A "is it null" column in front of each one puts them where PostgreSQL put them --
               rows with no distance (no point was given) after the ones that have it. */
            ListingSortOrder.Distance => (
                "CASE WHEN distance_metres IS NULL THEN 1 ELSE 0 END, distance_metres ASC, name ASC",
                "CASE WHEN p.distance_metres IS NULL THEN 1 ELSE 0 END, p.distance_metres ASC, p.name ASC"),
            ListingSortOrder.Name => (
                "name ASC",
                "p.name ASC"),
            _ => (
                "CASE WHEN relevance IS NULL THEN 1 ELSE 0 END, relevance DESC, CASE WHEN distance_metres IS NULL THEN 1 ELSE 0 END, distance_metres ASC, name ASC",
                "CASE WHEN p.relevance IS NULL THEN 1 ELSE 0 END, p.relevance DESC, CASE WHEN p.distance_metres IS NULL THEN 1 ELSE 0 END, p.distance_metres ASC, p.name ASC"),
        };

        /* With nothing typed there is no text matching at all: no full-text joins, a constant
           relevance, and no extra filter. With something typed, all three appear together. */
        var hasText = query.Text is not null;
        var textJoins = hasText
            ? @"LEFT JOIN CONTAINSTABLE(listings, name, @ftText) ftn ON ftn.[KEY] = li.id
              LEFT JOIN CONTAINSTABLE(listings, description, @ftText) ftd ON ftd.[KEY] = li.id
              CROSS APPLY hl_similarity('listing', li.id, @text) sim"
            : string.Empty;

        /* Preserve the previous search weighting: the name counts double the description.
   SQL Server full-text RANK is 0-1000, so it is scaled to sit beside a similarity
   score of 0-1 rather than swamping it. */
        var relevance = hasText
            ? "(ISNULL(ftn.[RANK], 0) * 2 + ISNULL(ftd.[RANK], 0)) / 1000.0 + sim.similarity"
            : "0";

        var textFilter = hasText
            ? @"AND (ftn.[KEY] IS NOT NULL
                     OR ftd.[KEY] IS NOT NULL
                     OR sim.similarity >= 0.3
                     OR EXISTS (SELECT 1
                                FROM listing_categories kc
                                JOIN categories kcat ON kcat.id = kc.category_id
                                LEFT JOIN category_search_terms kt ON kt.category_id = kcat.id
                                CROSS APPLY hl_similarity('category', kcat.id, @text) csim
                                WHERE kc.listing_id = li.id
                                  AND (csim.similarity >= 0.3 OR kt.term = @text)))"
            : string.Empty;

        await using var command = db.CreateCommand(
            string.Format(SearchSqlFormat, pageOrder, resultOrder, textJoins, relevance, textFilter));
        AddText(command, "text", query.Text);
        if (hasText)
        {
            AddText(command, "ftText", FullTextQuery(query.Text!));
        }
        AddText(command, "city", query.CitySlug);
        AddText(command, "locality", query.LocalitySlug);
        AddText(command, "category", query.CategorySlug);
        AddDouble(command, "lat", query.Latitude);
        AddDouble(command, "lng", query.Longitude);
        AddDouble(command, "radius", query.RadiusKm * 1000);
        command.Parameters.AddWithValue("pageSize", query.PageSize);
        command.Parameters.AddWithValue("offset", query.Offset);

        await using var reader = await command.ExecuteReaderAsync(ct);
        var items = new List<SearchListingDto>();
        var total = 0;
        while (await reader.ReadAsync(ct))
        {
            total = (int)reader.GetInt64(0);
            if (reader.IsDBNull(1))
            {
                continue;   // the placeholder row: this page holds no listings
            }

            var slug = reader.GetString(1);
            var citySlug = reader.GetString(3);
            var localitySlug = reader.GetString(4);
            var categorySlug = reader.GetString(6);
            items.Add(new SearchListingDto(
                slug,
                reader.GetString(2),
                citySlug,
                localitySlug,
                reader.GetString(5),
                categorySlug,
                reader.GetString(7),
                reader.IsDBNull(11) ? null : (int)Math.Round(reader.GetDouble(11)),
                reader.IsDBNull(8) ? null : reader.GetString(8),
                reader.GetString(9),
                reader.IsDBNull(10) ? null : reader.GetString(10),
                $"/{citySlug}/{localitySlug}/{categorySlug}/{slug}",
                reader.IsDBNull(12) ? null : storage.PublicUrl(reader.GetString(12)),
                reader.IsDBNull(13) ? null : storage.PublicUrl(reader.GetString(13))));
        }

        var totalPages = (int)Math.Ceiling(total / (double)query.PageSize);
        return Results.Ok(new SearchResultsDto(items, query.Page, query.PageSize, total, totalPages));
    }

    /// <summary>
    /// Turns what someone typed into a condition CONTAINSTABLE will accept.
    ///
    /// PostgreSQL had websearch_to_tsquery, which took any text at all and made sense of it. SQL
    /// Server has no such thing: a quote or a stray AND in the string is a syntax error rather than
    /// a search for that word. So each word is reduced to letters and digits, quoted, given a
    /// trailing * so "plumb" still finds "plumbing", and the words are joined with OR -- matching
    /// any of them, which is what a directory search should do.
    ///
    /// Returns a condition that matches nothing when the text had no usable word in it, so the
    /// caller never has to special-case that.
    /// </summary>
    public static string FullTextQuery(string text)
    {
        var words = text
            .Split((char[]?)null, StringSplitOptions.RemoveEmptyEntries)
            .Select(word => new string(word.Where(char.IsLetterOrDigit).ToArray()))
            .Where(word => word.Length > 0)
            .Take(10)   // a sentence pasted into the box should not become a hundred-term query
            .Select(word => $"\"{word}*\"")
            .ToArray();

        return words.Length == 0 ? "\"\"" : string.Join(" OR ", words);
    }

    private static void AddText(SqlCommand command, string name, string? value) =>
        command.Parameters.Add(new SqlParameter(name, SqlDbType.NVarChar) { Value = (object?)value ?? DBNull.Value });

    private static void AddDouble(SqlCommand command, string name, double? value) =>
        command.Parameters.Add(new SqlParameter(name, SqlDbType.Float) { Value = (object?)value ?? DBNull.Value });
}
